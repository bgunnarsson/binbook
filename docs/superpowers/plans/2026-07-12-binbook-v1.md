# Binbook v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build binbook v1 — a framework-agnostic component workshop with a working HTML adapter, proving the full loop (discover `.binbook.json` → adapter renders via `{{arg}}` interpolation → viewer displays in a sandboxed iframe → hot reload) plus a controls/args panel, viewport toggles, and a docs view.

**Architecture:** A pnpm monorepo with four packages under the `@bgunnarsson` scope. `binbook-core` (Node) discovers stories, resolves args, and runs a Vite dev server that mounts an API + websocket. `binbook-adapter-html` turns a component ref + resolved args into an HTML string. `binbook-viewer` (React + Tailwind) renders the sidebar/canvas/controls/docs and hosts each story in a sandboxed iframe via `srcdoc`. `binbook-types` holds the shared contracts. The canvas iframe is the isolation boundary that keeps the viewer's React runtime away from story markup.

**Tech Stack:** Node + Vite (core/dev server), React + Tailwind (viewer), TypeScript, zod (story validation), fast-glob (discovery), Vitest (unit/integration), Playwright (e2e), tsx (running TS in dev).

## Conventions (read once, apply to every task)

- **Line endings:** LF only. An `.editorconfig` enforces this; never introduce CRLF.
- **TDD:** every logic task writes the failing test first, watches it fail, then implements.
- **Commits:** small and frequent, conventional-commit style (`feat:`, `test:`, `chore:`). Commit messages contain no author/co-author trailers and no tooling attribution.
- **Package manager:** `pnpm` only. Run package scripts with `pnpm --filter <pkg> <script>`.
- **Imports between packages** use the published names (`@bgunnarsson/binbook-types`), resolved through the workspace, never deep relative paths across package boundaries.
- **No build step in v1** for `-types`, `-core`, `-adapter-html`: they run from TypeScript source via `tsx`/Vitest. Only `-viewer` is built by Vite. This keeps the loop fast; publishing is a later milestone.

```ts
export const ARG_TYPES = ["text", "select", "boolean", "number", "color"] as const;
export type ArgType = (typeof ARG_TYPES)[number];

export function isArgType(value: unknown): value is ArgType {
  return typeof value === "string" && (ARG_TYPES as readonly string[]).includes(value);
}

export interface ArgSpec {
  type: ArgType;
  default: unknown;
  options?: string[]; // for "select"
  label?: string;
}

export type ArgSchema = Record<string, ArgSpec>;

export interface Variant {
  name: string;
  args: Record<string, unknown>;
}

/** One parsed .binbook.json component, expanded per top-level metadata. */
export interface Story {
  /** Stable id: `${title}` slugified + variant handled separately. */
  id: string;
  title: string;              // e.g. "Components/Button"
  description?: string;
  component: string;          // adapter-interpreted render target
  args: ArgSchema;
  variants: Variant[];
  /** Absolute path to the source .binbook.json (for watch + resolving component). */
  file: string;
}

export interface RenderContext {
  story: Story;
  component: string;
  args: Record<string, unknown>;
  root: string;
}

export interface RenderResult {
  html: string;
  assets?: string[];
}

export interface DocsMeta {
  source?: string; // component source snippet
}

export type Dispose = () => void;

export interface Adapter {
  name: string;
  storyGlob: string;
  render(ctx: RenderContext): Promise<RenderResult>;
  watch?(story: Story, onChange: () => void): Dispose;
  docs?(story: Story): Promise<DocsMeta>;
}

export interface BinbookConfig {
  stories: string[];
  adapters: Adapter[];
  port?: number;
  staticDirs?: string[];
}

/** Nested tree the sidebar consumes. */
export interface TreeNode {
  name: string;
  path: string;               // full slash path
  children: TreeNode[];
  /** Present on leaf nodes that map to a story variant. */
  leaf?: { storyId: string; variant: string };
}
```

`packages/core/src/story-schema.ts`:
```ts
import { z } from "zod";

const argSpec = z
  .object({
    type: z.enum(["text", "select", "boolean", "number", "color"]),
    default: z.unknown(),
    options: z.array(z.string()).optional(),
    label: z.string().optional(),
  })
  .superRefine((val, ctx) => {
    if (val.type === "select" && (!val.options || val.options.length === 0)) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, message: "select arg requires non-empty `options`" });
    }
  });

export const storyFileSchema = z.object({
  title: z.string().min(1),
  description: z.string().optional(),
  component: z.string().min(1),
  args: z.record(argSpec).default({}),
  stories: z
    .array(z.object({ name: z.string().min(1), args: z.record(z.unknown()).default({}) }))
    .min(1),
});

export type StoryFile = z.infer<typeof storyFileSchema>;
```

`packages/core/src/parse-story.ts`:
```ts
import type { Story } from "@bgunnarsson/binbook-types";
import { storyFileSchema } from "./story-schema";

export function slugify(title: string): string {
  return title
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");
}

export function parseStoryFile(json: unknown, file: string): Story {
  const parsed = storyFileSchema.parse(json);
  return {
    id: slugify(parsed.title),
    title: parsed.title,
    description: parsed.description,
    component: parsed.component,
    args: parsed.args,
    variants: parsed.stories,
    file,
  };
}
```

```ts
import type { ArgSchema } from "@bgunnarsson/binbook-types";

export function resolveArgs(
  schema: ArgSchema,
  variantArgs: Record<string, unknown>,
  liveArgs: Record<string, unknown>,
): Record<string, unknown> {
  const out: Record<string, unknown> = {};
  for (const [key, spec] of Object.entries(schema)) {
    if (key in liveArgs) out[key] = liveArgs[key];
    else if (key in variantArgs) out[key] = variantArgs[key];
    else out[key] = spec.default;
  }
  return out;
}
```

```ts
import type { BinbookConfig, Story } from "@bgunnarsson/binbook-types";
import { parseStoryFile } from "./parse-story";

export async function discoverStories(config: Pick<BinbookConfig, "stories">): Promise<Story[]> {
  const files = await fg(config.stories, { absolute: true });
  const stories: Story[] = [];
  for (const file of files.sort()) {
    const json = JSON.parse(await readFile(file, "utf8"));
    stories.push(parseStoryFile(json, file));
  }
  return stories;
}
```

```ts
export function buildTree(stories: Story[]): TreeNode[] {
  const roots: TreeNode[] = [];

  const findOrCreate = (siblings: TreeNode[], name: string, path: string): TreeNode => {
    let node = siblings.find((n) => n.name === name && !n.leaf);
    if (!node) {
      node = { name, path, children: [] };
      siblings.push(node);
    }
    return node;
  };

  for (const story of stories) {
    const segments = story.title.split("/").map((s) => s.trim());
    let cursor = roots;
    let acc = "";
    for (const seg of segments) {
      acc = acc ? `${acc}/${seg}` : seg;
      const node = findOrCreate(cursor, seg, acc);
      cursor = node.children;
    }
    for (const variant of story.variants) {
      cursor.push({
        name: variant.name,
        path: `${story.title}/${variant.name}`,
        children: [],
        leaf: { storyId: story.id, variant: variant.name },
      });
    }
  }
  return roots;
}
```

### Task 7: `{{arg}}` interpolation engine

The interpolation rules (from the spec):
- `{{name}}` → the arg value, HTML-escaped.
- A `{{name}}` standing where a **boolean attribute** would go: if the value is `true`, render the bare attribute name; if `false`, render nothing. We detect this by the placeholder being the *entire* token between whitespace/`>` boundaries whose arg resolves to a boolean.

```ts
import { escapeHtml } from "./escape";

const TOKEN = /\{\{\s*([a-zA-Z0-9_]+)\s*\}\}/g;

export function interpolate(template: string, args: Record<string, unknown>): string {
  return template.replace(TOKEN, (_match, name: string) => {
    const value = args[name];
    if (value === undefined || value === null) return "";
    if (typeof value === "boolean") {
      // boolean attribute semantics: the attribute NAME is the arg key
      return value ? name : "";
    }
    return escapeHtml(String(value));
  });
}
```

Note: for boolean args, the emitted bare attribute is the arg key itself (`disabled`), matching the design's `{{disabled}}` example. This keeps authoring to a single placeholder.

```ts
import { readFile } from "node:fs/promises";
import { watch } from "node:fs";
import path from "node:path";
import type { Adapter, DocsMeta, RenderContext, RenderResult, Story } from "@bgunnarsson/binbook-types";
import { interpolate } from "./interpolate";

function componentPath(story: Story, component: string): string {
  // component is relative to the story file's directory
  return path.resolve(path.dirname(story.file), component);
}

export function html(): Adapter {
  return {
    name: "html",
    storyGlob: "**/*.binbook.json",

    async render(ctx: RenderContext): Promise<RenderResult> {
      const file = componentPath(ctx.story, ctx.component);
      const template = await readFile(file, "utf8");
      return { html: interpolate(template, ctx.args) };
    },

    watch(story: Story, onChange: () => void) {
      const file = componentPath(story, story.component);
      const watcher = watch(file, () => onChange());
      const storyWatcher = watch(story.file, () => onChange());
      return () => {
        watcher.close();
        storyWatcher.close();
      };
    },

    async docs(story: Story): Promise<DocsMeta> {
      const file = componentPath(story, story.component);
      return { source: await readFile(file, "utf8") };
    },
  };
}

export default html;
```

```ts
import type { Adapter, Story } from "@bgunnarsson/binbook-types";

export class AdapterRegistry {
  constructor(private readonly adapters: Adapter[]) {
    if (adapters.length === 0) throw new Error("No adapters registered");

  forStory(story: Story): Adapter {

  }
```

```ts
export async function renderStory(
```

```ts
import type { Connect, Plugin, ViteDevServer } from "vite";
import { WebSocketServer } from "ws";
import type { BinbookConfig, RenderResult, Story, TreeNode } from "@bgunnarsson/binbook-types";
import { discoverStories } from "./discover";
import { buildTree } from "./tree";
import { AdapterRegistry } from "./registry";
import { renderStory } from "./render";

export interface ApiHandlers {
  stories(): Promise<{ tree: TreeNode[]; stories: Story[] }>;
  render(body: { storyId: string; variant: string; args: Record<string, unknown> }): Promise<RenderResult>;
  reload(): Promise<void>;
}

export async function createApiHandlers(opts: {
  root: string;
  config: BinbookConfig;
}): Promise<ApiHandlers> {
  const registry = new AdapterRegistry(opts.config.adapters);
  let stories = await discoverStories(opts.config);

  return {
    async stories() {
      return { tree: buildTree(stories), stories };
    },
    async render({ storyId, variant, args }) {
      const story = stories.find((s) => s.id === storyId);
      if (!story) throw new Error(`Unknown story ${storyId}`);
      return renderStory(registry, story, variant, args ?? {}, opts.root);
    },
    async reload() {
      stories = await discoverStories(opts.config);
    },
  };
}

/** JSON body reader for connect middleware. */
function readJson(req: Connect.IncomingMessage): Promise<any> {
  return new Promise((resolve, reject) => {
    let data = "";
    req.on("data", (c) => (data += c));
    req.on("end", () => resolve(data ? JSON.parse(data) : {}));
    req.on("error", reject);
  });
}

export function binbookServer(opts: { root: string; config: BinbookConfig }): Plugin {
  return {
    name: "binbook-server",
    async configureServer(server: ViteDevServer) {
      const api = await createApiHandlers(opts);
      const wss = new WebSocketServer({ noServer: true });

      server.httpServer?.on("upgrade", (req, socket, head) => {
        if (req.url === "/__binbook_ws") {
          wss.handleUpgrade(req, socket, head, (ws) => wss.emit("connection", ws));
        }
      });

      const pushReload = () => wss.clients.forEach((c) => c.readyState === 1 && c.send("reload"));

      // Watch story files + component files; re-discover + push reload.
      for (const story of (await api.stories()).stories) {
        const adapter = opts.config.adapters[0];
        adapter.watch?.(story, async () => {
          await api.reload();
          pushReload();
        });
      }

      server.middlewares.use("/api/stories", async (_req, res) => {
        res.setHeader("content-type", "application/json");
        res.end(JSON.stringify(await api.stories()));
      });

      server.middlewares.use("/api/render", async (req, res) => {
        try {
          const body = await readJson(req);
          const result = await api.render(body);
          res.setHeader("content-type", "application/json");
          res.end(JSON.stringify(result));
        } catch (err) {
          res.statusCode = 400;
          res.end(JSON.stringify({ error: (err as Error).message }));
        }
      });
    },
  };
}

/** Programmatic dev server used by the CLI. */
export async function createDevServer(opts: {
  root: string;
  viewerRoot: string;
  config: BinbookConfig;
}) {
  const { createServer } = await import("vite");
  return createServer({
    root: opts.viewerRoot,
    server: { port: opts.config.port ?? 6006 },
    plugins: [binbookServer({ root: opts.root, config: opts.config })],
  });
}
```

```ts
export function fuzzyMatch(query: string, target: string): boolean {
  const q = query.toLowerCase();
  const t = target.toLowerCase();
```
